\documentclass[border=4pt]{standalone}
\usepackage{manafig}
\renewcommand{\sfdefault}{qhv}
\def\sq{0.27}   % square pitch
\def\sqs{0.22}  % square size
% a stage: #1 = row index, #2 = candidates, #3 = survivors, #4 = games each, #5 = bar length (cm)
\newcommand{\stage}[5]{%
  \pgfmathsetmacro{\yy}{-(#1-1)*1.75}
  \foreach \i in {0,...,\numexpr#2-1\relax}{
    \pgfmathtruncatemacro{\col}{int(\i/4)}
    \pgfmathtruncatemacro{\row}{mod(\i,4)}
    \ifnum\i<#3
      \fill[oursc,rounded corners=1pt] (3.0+\col*\sq,\yy-\row*\sq) rectangle (3.0+\col*\sq+\sqs,\yy-\row*\sq-\sqs);
    \else
      \fill[manaN!70!white,rounded corners=1pt] (3.0+\col*\sq,\yy-\row*\sq) rectangle (3.0+\col*\sq+\sqs,\yy-\row*\sq-\sqs);
    \fi
  }
  \node[anchor=west,font=\sffamily\bfseries\Large,text=ink] at (0,\yy-0.4) {#2};
  \node[anchor=west,font=\sffamily\small,text=inksoft] at (0,\yy-0.88) {candidates};
  % games per candidate bar
  \fill[oursc!85,rounded corners=1.5pt] (9.7,\yy-0.25) rectangle (9.7+#5,\yy-0.65);
  \node[anchor=west,font=\sffamily\bfseries\normalsize,text=ink] at (9.7+#5+0.12,\yy-0.45) {#4};
  \node[anchor=west,font=\sffamily\small,text=inksoft] at (9.7,\yy-1.0) {games each};
}
