% weight strip: sixteen squares, #1 = weight, centred at height #2
\newcommand{\wstrip}[2]{%
  \foreach \i in {0,...,15}{
    \pgfmathsetmacro{\fr}{max(0,min(1,#1-\i))}
    \draw[gridc,fill=white,line width=0.4pt] (3.55+\i*0.205,#2-0.09) rectangle (3.55+\i*0.205+0.18,#2+0.09);
    \ifdim\fr pt>0pt
      \fill[ink!75] (3.55+\i*0.205,#2-0.09) rectangle (3.55+\i*0.205+\fr*0.18,#2+0.09);
    \fi
  }}
% one row: #1 top y, #2 weight, #3 name, #4 value A, #5 value B
\newcommand{\scorerow}[5]{%
  \pgfmathsetmacro{\h}{0.7+0.3*#2}
  \pgfmathsetmacro{\t}{max(0.3,0.2+0.06*#2)}
  \pgfmathsetmacro{\yc}{#1-\h/2}
  \draw[gridc,line width=0.4pt] (0,#1) -- (\xo+6.6,#1);
  \node[anchor=west,font=\sffamily\bfseries\normalsize,text=ink] at (0,\yc+0.14) {#3};
  \node[anchor=west,font=\sffamily\normalsize,text=inksoft] at (0,\yc-0.17) {counts #2 of 16};
  \wstrip{#2}{\yc}
  \pgfmathsetmacro{\ya}{\yc+\t/2+0.02}
  \pgfmathsetmacro{\yb}{\yc-\t/2-0.02}
  \fill[manaN,rounded corners=2.5pt] (\xo,\ya-\t/2) rectangle (\xo+#4*\us,\ya+\t/2);
  \fill[oursc,rounded corners=2.5pt] (\xo,\yb-\t/2) rectangle (\xo+#5*\us,\yb+\t/2);
  \node[anchor=west,font=\sffamily\bfseries\normalsize,text=inksoft] at (\xo+#4*\us+0.1,\ya) {#4};
  \node[anchor=west,font=\sffamily\bfseries\normalsize,text=oursc] at (\xo+#5*\us+0.1,\yb) {#5};
}
