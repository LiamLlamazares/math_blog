% chip: x y name(two lines) colour1 colour2 (second may be empty -> same pip count 1)
\newcommand{\landchip}[5]{%
  \draw[fill=white,draw=ink!50,line width=0.5pt,rounded corners=4pt] (#1-1.05,#2-0.72) rectangle (#1+1.05,#2+0.72);
  \node[font=\sffamily\bfseries\small,text=ink,align=center,inner sep=0pt] at (#1,#2+0.28) {#3};
  \ifx\relax#5\relax\spip{#1}{#2-0.38}{#4}\else\spip{#1-0.3}{#2-0.38}{#4}\spip{#1+0.3}{#2-0.38}{#5}\fi}
