% \lt{x}{y}{count}{name}{c1}{c2}{c3}: land token, split into two or three colours (c3 = - for two); gold outline marks the new land
\newcommand{\lt}[6]{%
  \edef\tca{\mfill{#5}}\edef\tcb{\mfill{#6}}%
  \node[landtok,anchor=north west,minimum width=4.5cm,minimum height=0.6cm,align=left,fill=white,font=\sffamily\bfseries\small,
    path picture={\fill[\tca] (path picture bounding box.south west) rectangle (path picture bounding box.north);
                  \fill[\tcb] (path picture bounding box.south) rectangle (path picture bounding box.north east);}]
    at (#1,#2) {\hspace{0.15cm}#3\,$\times$\,#4};}
\newcommand{\ltt}[4]{% three colours: x y count name  (water, dark, fire)
  \node[landtok,anchor=north west,minimum width=4.5cm,minimum height=0.6cm,align=left,fill=white,font=\sffamily\bfseries\small,draw=newc,line width=1.8pt,
    path picture={\fill[manaUp] (path picture bounding box.south west) rectangle ($(path picture bounding box.south west)!0.3333!(path picture bounding box.south east)+(0,0.6)$);
                  \fill[manaBp] ($(path picture bounding box.south west)!0.3333!(path picture bounding box.south east)$) rectangle ($(path picture bounding box.south west)!0.6667!(path picture bounding box.south east)+(0,0.6)$);
                  \fill[manaRp] ($(path picture bounding box.south west)!0.6667!(path picture bounding box.south east)$) rectangle (path picture bounding box.north east);}]
    at (#1,#2) {\hspace{0.15cm}#3\,$\times$\,#4};}
